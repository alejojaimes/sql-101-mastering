    %% ============================================
    %% GEO MODULE
    %% ============================================

    DEPARTMENTS ||--o{ MUNICIPALITIES : "contains"
    MUNICIPALITIES ||--o{ FARMERS : "lives_in"
    MUNICIPALITIES ||--o{ FARMS : "includes"
    MUNICIPALITIES ||--o{ COLLECTION_POINTS : "hosts"

    DEPARTMENTS {
        varchar department_code PK
        varchar department_name "UNIQUE, NOT NULL"
    }

    MUNICIPALITIES {
        varchar municipality_code PK
        varchar municipality_name "NOT NULL"
        varchar department_code FK "NOT NULL"
    }

    %% ============================================
    %% FARM MODULE
    %% ============================================

    FARMERS ||--o{ FARMS : "owns"
    FARMS ||--o{ PLOTS : "divided_into"
    PLOTS ||--o{ PLOT_VARIETIES : "grows"
    COFFEE_VARIETIES ||--o{ PLOT_VARIETIES : "planted_in"
    FARMS ||--o{ FARM_CERTIFICATIONS : "holds"
    CERTIFICATIONS ||--o{ FARM_CERTIFICATIONS : "granted_to"

    FARMERS {
        int farmer_id PK
        varchar document_number "UNIQUE, NOT NULL"
        varchar first_name "NOT NULL"
        varchar last_name "NOT NULL"
        date birth_date "NOT NULL"
        char gender "CHECK(M,F,O), NOT NULL"
        varchar phone_number "NOT NULL"
        varchar email "UNIQUE, NULL"
        date membership_date "DEFAULT CURRENT_DATE"
        boolean is_active "DEFAULT TRUE"
        varchar municipality_code FK "NOT NULL"
    }

    FARMS {
        int farm_id PK
        int farmer_id FK "NOT NULL"
        varchar municipality_code FK "NOT NULL"
        varchar farm_name "UNIQUE(farmer_id, farm_name), NOT NULL"
        varchar rural_district "NOT NULL"
        numeric total_area_ha "CHECK(> 0), NOT NULL"
        int altitude_masl "NOT NULL"
        numeric latitude "NULL"
        numeric longitude "NULL"
        timestamp registered_at "DEFAULT CURRENT_TIMESTAMP"
    }

    PLOTS {
        int plot_id PK
        int farm_id FK "NOT NULL"
        varchar plot_code "UNIQUE(farm_id, plot_code), NOT NULL"
        numeric area_ha "CHECK(> 0), NOT NULL"
        smallint planting_year "NOT NULL"
        varchar shade_type "CHECK(full_sun,partial_shade,full_shade)"
    }

    COFFEE_VARIETIES {
        int variety_id PK
        varchar variety_name "UNIQUE, NOT NULL"
        varchar species "CHECK(arabica,robusta), NOT NULL"
        boolean is_rust_resistant "DEFAULT FALSE"
        text description "NULL"
    }

    PLOT_VARIETIES {
        int plot_variety_id PK
        int plot_id FK "UNIQUE(plot_id, variety_id), NOT NULL"
        int variety_id FK "NOT NULL"
        numeric planted_percentage "CHECK(> 0 AND <= 100), NOT NULL"
        int tree_count "NOT NULL"
    }

    CERTIFICATIONS {
        int certification_id PK
        varchar certification_name "UNIQUE, NOT NULL"
        varchar issuing_body "NOT NULL"
        numeric premium_per_kg "CHECK(>= 0), DEFAULT 0"
    }

    FARM_CERTIFICATIONS {
        int farm_certification_id PK
        int farm_id FK "UNIQUE(farm, certification, start_date)"
        int certification_id FK "NOT NULL"
        date start_date "NOT NULL"
        date end_date "CHECK(> start_date), NULL"
        varchar certificate_number "UNIQUE, NOT NULL"
    }

    %% ============================================
    %% BUY MODULE
    %% ============================================

    COLLECTION_POINTS ||--o{ DAILY_PRICES : "publishes"
    COLLECTION_POINTS ||--o{ DELIVERIES : "receives"
    FARMS ||--o{ DELIVERIES : "makes"
    DELIVERIES ||--o| QUALITY_ANALYSES : "analyzed_by"

    COLLECTION_POINTS {
        int collection_point_id PK
        varchar point_name "UNIQUE, NOT NULL"
        varchar municipality_code FK "NOT NULL"
        varchar address_line "NOT NULL"
        varchar phone_number "NULL"
        boolean is_active "DEFAULT TRUE"
    }

    DAILY_PRICES {
        int daily_price_id PK
        int collection_point_id FK "UNIQUE(point, date), NOT NULL"
        date price_date "NOT NULL"
        numeric price_per_kg "CHECK(> 0), NOT NULL"
        numeric reference_yield_factor "DEFAULT 94"
    }

    DELIVERIES {
        varchar delivery_id PK "DLV-YYYYMMDD-NNNNNN"
        int farm_id FK "NOT NULL"
        int collection_point_id FK "NOT NULL"
        timestamp delivered_at "DEFAULT CURRENT_TIMESTAMP"
        varchar coffee_type "CHECK(dry_parchment,wet_parchment,cherry)"
        smallint bag_count "CHECK(> 0), NOT NULL"
        numeric gross_weight_kg "CHECK(> 0), NOT NULL"
        varchar status "CHECK(received,accepted,rejected,settled)"
    }

    QUALITY_ANALYSES {
        varchar analysis_id PK "QA-YYYYMMDD-NNNNNN"
        varchar delivery_id FK "UNIQUE, NOT NULL"
        timestamp analyzed_at "DEFAULT CURRENT_TIMESTAMP"
        numeric moisture_pct "CHECK(0-100), NOT NULL"
        numeric yield_factor "NOT NULL"
        numeric defects_pct "DEFAULT 0"
        numeric cup_score "NULL"
        boolean is_accepted "NOT NULL"
        varchar analyst_name "NOT NULL"
        text observations "NULL"
    }

    %% ============================================
    %% PAY MODULE
    %% ============================================

    DELIVERIES ||--o| SETTLEMENTS : "settled_by"
    DAILY_PRICES ||--o{ SETTLEMENTS : "applied_in"
    SETTLEMENTS ||--o{ PAYMENTS : "paid_by"

    SETTLEMENTS {
        varchar settlement_id PK "STL-YYYYMMDD-NNNNNN"
        varchar delivery_id FK "UNIQUE, NOT NULL"
        int daily_price_id FK "NOT NULL"
        numeric net_weight_kg "CHECK(> 0), NOT NULL"
        numeric price_per_kg_applied "CHECK(> 0), NOT NULL"
        numeric deductions_amount "CHECK(>= 0), DEFAULT 0"
        numeric gross_amount "GENERATED (weight * price)"
        numeric net_amount "GENERATED (gross - deductions)"
        timestamp settled_at "DEFAULT CURRENT_TIMESTAMP"
        varchar status "CHECK(pending,partially_paid,paid,cancelled)"
    }

    PAYMENTS {
        varchar payment_id PK "PAY-YYYYMMDD-NNNNNN"
        varchar settlement_id FK "NOT NULL"
        varchar payment_method "CHECK(cash,bank_transfer,check)"
        numeric amount "CHECK(> 0), NOT NULL"
        timestamp paid_at "DEFAULT CURRENT_TIMESTAMP"
        varchar transaction_reference "UNIQUE, NULL"
    }
